네, GitHub 연동 없이 **내 컴퓨터(로컬)에 있는 JSON 파일**을 읽고 쓰는 방식으로 구조를 변경해 드리겠습니다. 이렇게 하면 설정이 훨씬 간단해지고 속도도 빨라집니다.

기존의 `utils_github.py` 대신 사용할 **`utils_local.py`**를 만들고, 메인 코드인 `app.py`를 살짝 수정하면 됩니다.

### 1단계: `utils_local.py` 생성

프로젝트 폴더에 `utils_local.py` 파일을 새로 만들고 아래 코드를 붙여넣으세요.
(기존 `utils_github.py`는 이제 사용하지 않습니다.)

```python
import json
import os
import streamlit as st

# 데이터가 저장될 폴더 (현재 폴더에 'data'라는 폴더를 만들어 관리하면 깔끔함)
DATA_DIR = "data"

# 데이터 폴더가 없으면 생성
if not os.path.exists(DATA_DIR):
    os.makedirs(DATA_DIR)

def get_file_path(filename):
    """파일의 전체 경로를 반환합니다."""
    return os.path.join(DATA_DIR, filename)

def load_json(filename):
    """로컬 파일에서 JSON 데이터를 읽어옵니다."""
    file_path = get_file_path(filename)
    
    if not os.path.exists(file_path):
        return {}
    
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            return json.load(f)
    except json.JSONDecodeError:
        return {}
    except Exception as e:
        st.error(f"파일 읽기 오류: {e}")
        return {}

def save_json(filename, data, message=None):
    """로컬 파일에 JSON 데이터를 저장합니다. (message 인자는 호환성을 위해 남겨둠)"""
    file_path = get_file_path(filename)
    
    try:
        with open(file_path, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=4, ensure_ascii=False)
        # 로컬이라 저장 성공 메시지는 너무 자주 뜨면 귀찮으므로 주석 처리하거나 필요 시 사용
        # st.success(f"{filename} 저장 완료!") 
    except Exception as e:
        st.error(f"파일 저장 오류: {e}")
```

### 2단계: `app.py` 수정

`app.py`에서 **GitHub 관련 코드를 제거**하고, 방금 만든 `utils_local`을 사용하도록 변경합니다.

```python
import streamlit as st
import pandas as pd
import plotly.express as px # 설치 안 했으면 pip install plotly 필요
from datetime import datetime
import traceback

# [변경] GitHub 모듈 대신 로컬 모듈 임포트
import utils_local as db  
import utils_logic as logic

# 1. 페이지 설정
st.set_page_config(page_title="나만의 뉴스룸 (Local)", page_icon="🏠", layout="wide")

# [변경] 필수 설정 확인 (GitHub 관련 키 제거)
if "GEMINI_API_KEY" not in st.secrets:
    st.error("🚨 .streamlit/secrets.toml에 GEMINI_API_KEY가 없습니다.")
    st.stop()

# 2. 사이드바 메뉴
menu = st.sidebar.selectbox("메뉴 선택", ["📰 뉴스룸 (메인)", "⚙️ 관리 대시보드"])

# ==========================================
# 메인 화면: 뉴스룸
# ==========================================
if menu == "📰 뉴스룸 (메인)":
    st.title("📰 1인 IT 뉴스룸 (로컬 모드)")
    st.caption("내 컴퓨터에 저장된 데이터로 브리핑을 보여줍니다.")

    # [변경] git.load_json -> db.load_json
    news_data = db.load_json("news_data.json")
    
    today = datetime.now().strftime("%Y-%m-%d")
    dates = sorted(news_data.keys(), reverse=True) if news_data else [today]
    
    if not news_data:
        st.info("📭 저장된 뉴스가 없습니다. 대시보드에서 수집해주세요.")
    else:
        selected_date = st.selectbox("날짜 선택", dates)
        
        if selected_date in news_data:
            content = news_data[selected_date]
            
            col1, col2 = st.columns([2, 1])
            with col1:
                st.markdown(content.get('analysis', '분석 내용 없음'))
            with col2:
                st.subheader("🔗 원본 링크")
                for item in content.get('raw_news', [])[:10]:
                    st.markdown(f"- [{item['title']}]({item['link']})")

            # 통계 (로컬 파일 저장)
            if "visited" not in st.session_state:
                stats = db.load_json("stats.json")
                stats["total_visits"] = stats.get("total_visits", 0) + 1
                db.save_json("stats.json", stats)
                st.session_state["visited"] = True

# ==========================================
# 관리 대시보드
# ==========================================
elif menu == "⚙️ 관리 대시보드":
    st.title("⚙️ 관리 대시보드")
    
    tab1, tab2, tab3 = st.tabs(["RSS 관리", "수집 및 분석", "시스템 상태"])
    
    # 1. RSS 관리
    with tab1:
        st.subheader("RSS 피드 목록")
        feeds_data = db.load_json("feeds.json")
        current_feeds = feeds_data.get("urls", [])
        
        new_feed = st.text_input("새 RSS URL 추가")
        if st.button("추가"):
            if new_feed and new_feed not in current_feeds:
                current_feeds.append(new_feed)
                db.save_json("feeds.json", {"urls": current_feeds})
                st.rerun()
        
        st.write("---")
        for url in current_feeds:
            c1, c2 = st.columns([4, 1])
            c1.text(url)
            if c2.button("삭제", key=url):
                current_feeds.remove(url)
                db.save_json("feeds.json", {"urls": current_feeds})
                st.rerun()

    # 2. 수집 및 분석
    with tab2:
        st.subheader("뉴스 수집 및 AI 분석")
        if st.button("🚀 실행 (Start)"):
            try:
                feeds_data = db.load_json("feeds.json")
                urls = feeds_data.get("urls", [])
                
                if not urls:
                    st.error("RSS 피드가 없습니다.")
                else:
                    with st.spinner("뉴스 수집 중..."):
                        raw_news = logic.fetch_rss_feeds(urls)
                    
                    with st.spinner("Gemini 분석 중..."):
                        analysis_result = logic.analyze_with_gemini(raw_news)
                    
                    with st.spinner("저장 중..."):
                        today_str = datetime.now().strftime("%Y-%m-%d")
                        full_data = db.load_json("news_data.json")
                        
                        full_data[today_str] = {
                            "analysis": analysis_result,
                            "raw_news": raw_news,
                            "updated_at": datetime.now().strftime("%H:%M:%S")
                        }
                        # 로컬 저장
                        db.save_json("news_data.json", full_data)
                    
                    st.success("완료되었습니다!")
                    
            except Exception as e:
                st.error("오류 발생")
                st.code(traceback.format_exc())

    # 3. 시스템 상태
    with tab3:
        stats = db.load_json("stats.json")
        st.metric("총 실행 횟수", stats.get("total_visits", 0))
        st.success("로컬 저장소: 연결됨 (./data 폴더)")
        if "GEMINI_API_KEY" in st.secrets:
            st.success("Gemini API: 연결됨")
```
